cmake_minimum_required(VERSION 3.16)
project(baby_english_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
add_compile_options(-Wall -Wextra -Werror -fsanitize=undefined -fno-sanitize-recover=undefined)
add_link_options(-fsanitize=undefined)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(core ${ROOT}/main/ep_core.c ${ROOT}/main/content.c ${ROOT}/main/ep_input.c ${ROOT}/main/ep_voice.c ${ROOT}/main/ep_opus.c ${ROOT}/main/ep_notices.c)
target_include_directories(core PUBLIC ${ROOT}/main)
foreach(name core input voice opus word_migration)
 add_executable(test_${name} test_${name}.c)
 target_link_libraries(test_${name} core)
 add_test(NAME ${name} COMMAND test_${name})
endforeach()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/host/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/managed_components/lvgl__lvgl" "${CMAKE_BINARY_DIR}/lvgl")
file(GLOB FONTS "${ROOT}/main/ep_font*.c")
add_executable(test_ui host/test_ui.c ${ROOT}/main/ui.c ${ROOT}/main/home_art.c ${ROOT}/main/ui_art.c ${ROOT}/main/word_art.c ${FONTS})
target_link_libraries(test_ui core lvgl m)
add_test(NAME native_ui COMMAND test_ui ${CMAKE_BINARY_DIR}/screens)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME curated_content COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_curated.py)
